#include <iostream>
#include <vector>

// Merges two sorted subarrays back into the original vector
void merge(std::vector<int>& arr, int left, int mid, int right) {
    // Temporary vector to store the sorted result
    std::vector<int> temp;
    
    int i = left;      // Starting index for the left subarray
    int j = mid + 1;   // Starting index for the right subarray

    // Compare elements from both halves and push the smaller one into temp
    while (i <= mid && j <= right) {
        if (arr[i] <= arr[j]) {
            temp.push_back(arr[i]);
            i++;
        } else {
            temp.push_back(arr[j]);
            j++;
        }
    }

    // Copy any remaining elements from the left subarray
    while (i <= mid) {
        temp.push_back(arr[i]);
        i++;
    }

    // Copy any remaining elements from the right subarray
    while (j <= right) {
        temp.push_back(arr[j]);
        j++;
    }

    // Copy the sorted elements from temp back into the original vector
    for (int k = left; k <= right; k++) {
        arr[k] = temp[k - left];
    }
}

// Main recursive function implementing the Merge Sort algorithm
void mergeSort(std::vector<int>& arr, int left, int right) {
    // Base case: If the subarray has 1 or 0 elements, it is already sorted
    if (left >= right) {
        return;
    }

    // Calculate the midpoint carefully to avoid integer overflow
    int mid = left + (right - left) / 2;

    // Recursively sort the first and second halves
    mergeSort(arr, left, mid);
    mergeSort(arr, mid + 1, right);

    // Merge the sorted halves
    merge(arr, left, mid, right);
}

// Helper function to print vector contents
void printVector(const std::vector<int>& arr) {
    for (int num : arr) {
        std::cout << num << " ";
    }
    std::cout << "\n";
}

int main() {
    std::vector<int> data = {38, 27, 43, 3, 9, 82, 10};

    std::cout << "Original vector: ";
    printVector(data);

    // Start sorting from the 0th index to the last element index
    mergeSort(data, 0, data.size() - 1);

    std::cout << "Sorted vector:   ";
    printVector(data);

    return 0;
}
